> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Nueva sección "Reportes" en el sidebar con vista Detalle de Ventas: filtros, paginación server-side, click en fila abre VentaDetalleModal, exportación CSV/XLSX.

**Architecture:** IPC handler `reportes.ts` ejecuta SQL dinámico con `db.$client` (mejor-sqlite3 subyacente) y retorna páginas de ventas `confirmadas`. El export reutiliza la infraestructura `exportar.ts` agregando la entidad `reporte_ventas`. La vista React conecta a ambos canales; para el detalle de una venta reutiliza `VentaDetalleModal` existente en `views/caja/`.

**Tech Stack:** TypeScript, React, Electron IPC, better-sqlite3, Drizzle ORM (solo para tipos), Vitest, @testing-library/react.

// packages/client/src/main/ipc/reportes.ts
import type { IpcMain } from 'electron';
import type { BetterSQLite3Database } from 'drizzle-orm/better-sqlite3';
import type Database from 'better-sqlite3';
import * as schema from '../db/schema';
import { getSession } from '../auth/session';

type Db = BetterSQLite3Database<typeof schema>;
type SqliteDb = Database.Database;

export interface ReporteVentasFiltros {
  fechaDesde?: Date;
  fechaHasta?: Date;
  clienteId?: string;
  categoriaId?: string;
  productoTexto?: string;
  usuarioId?: string;
  medioPago?: string;
}

export interface ReporteVentasRow {
  id: string;
  numero_interno: string;
  fecha: number;
  cliente_razon_social: string;
  vendedor_nombre: string;
  medios_pago: string;
  subtotal: number;
  importe_iva: number;
  total: number;
}

export interface ReporteVentasResult {
  rows: ReporteVentasRow[];
  total: number;
  totalImporte: number;
  page: number;
  totalPages: number;
}

export interface VentaDetalleExportRow {
  numero_interno: string;
  fecha: number;
  cliente_razon_social: string;
  vendedor_nombre: string;
  sku: string;
  descripcion: string;
  categoria_nombre: string;
  cantidad: number;
  precio_unitario: number;
  descuento_linea: number;
  alicuota_iva: number;
  subtotal_linea: number;
  total_venta: number;
}

function getSqlite(db: Db): SqliteDb {
  return (db as any).$client as SqliteDb;
}

function buildWhere(filtros: ReporteVentasFiltros): { where: string; params: unknown[] } {
  const conds: string[] = ["v.estado = 'confirmada'"];
  const params: unknown[] = [];

  if (filtros.fechaDesde) {
    conds.push('v.fecha >= ?');
    params.push(Math.floor(filtros.fechaDesde.getTime() / 1000));
  }
  if (filtros.fechaHasta) {
    const d = new Date(filtros.fechaHasta);
    d.setHours(23, 59, 59, 999);
    conds.push('v.fecha <= ?');
    params.push(Math.floor(d.getTime() / 1000));
  }
  if (filtros.clienteId) {
    conds.push('v.cliente_id = ?');
    params.push(filtros.clienteId);
  }
  if (filtros.usuarioId) {
    conds.push('v.usuario_id = ?');
    params.push(filtros.usuarioId);
  }
  if (filtros.medioPago) {
    conds.push('EXISTS (SELECT 1 FROM venta_pago vp2 WHERE vp2.venta_id = v.id AND vp2.medio_pago = ?)');
    params.push(filtros.medioPago);
  }
  if (filtros.categoriaId || filtros.productoTexto) {
    let sub = `EXISTS (
      SELECT 1 FROM venta_detalle vd
      JOIN producto p ON p.id = vd.producto_id
      WHERE vd.venta_id = v.id`;
    if (filtros.categoriaId) {
      sub += ' AND p.categoria_id = ?';
      params.push(filtros.categoriaId);
    }
    if (filtros.productoTexto) {
      const like = `%${filtros.productoTexto}%`;
      sub += ' AND (p.sku LIKE ? OR p.nombre_corto LIKE ? OR vd.descripcion LIKE ?)';
      params.push(like, like, like);
    }
    sub += ')';
    conds.push(sub);
  }

  return { where: conds.join(' AND '), params };
}

const CABECERA_SQL = `
  SELECT
    v.id, v.numero_interno, v.fecha,
    c.razon_social  AS cliente_razon_social,
    u.nombre_completo AS vendedor_nombre,
    COALESCE(GROUP_CONCAT(DISTINCT vp.medio_pago), v.forma_pago) AS medios_pago,
    v.subtotal, v.importe_iva, v.total
  FROM venta v
  JOIN cliente c ON c.id = v.cliente_id
  JOIN usuario u ON u.id = v.usuario_id
  LEFT JOIN venta_pago vp ON vp.venta_id = v.id
  WHERE `;

const CABECERA_GROUP = `
  GROUP BY v.id
  ORDER BY v.fecha DESC, v.numero_interno DESC`;

export function detalleVentasHandler(
  db: Db,
  filtros: ReporteVentasFiltros,
  page: number,
  pageSize: number
): ReporteVentasResult {
  const sqlite = getSqlite(db);
  const { where, params } = buildWhere(filtros);

  const count = sqlite.prepare(`
    SELECT COUNT(*) AS total, COALESCE(SUM(v.total), 0) AS totalImporte
    FROM venta v
    JOIN cliente c ON c.id = v.cliente_id
    JOIN usuario u ON u.id = v.usuario_id
    WHERE ${where}
  `).get(params) as { total: number; totalImporte: number };

  const offset = (page - 1) * pageSize;
  const rows = sqlite.prepare(
    CABECERA_SQL + where + CABECERA_GROUP + '\n  LIMIT ? OFFSET ?'
  ).all([...params, pageSize, offset]) as ReporteVentasRow[];

  return {
    rows,
    total: count.total,
    totalImporte: count.totalImporte,
    page,
    totalPages: Math.max(1, Math.ceil(count.total / pageSize)),
  };
}

export function reporteVentasQueryAll(db: Db, filtros: ReporteVentasFiltros): ReporteVentasRow[] {
  const sqlite = getSqlite(db);
  const { where, params } = buildWhere(filtros);
  return sqlite.prepare(CABECERA_SQL + where + CABECERA_GROUP).all(params) as ReporteVentasRow[];
}

export function reporteVentasDetalleQueryAll(db: Db, filtros: ReporteVentasFiltros): VentaDetalleExportRow[] {
  const sqlite = getSqlite(db);
  const { where, params } = buildWhere(filtros);
  return sqlite.prepare(`
    SELECT
      v.numero_interno, v.fecha,
      c.razon_social   AS cliente_razon_social,
      u.nombre_completo AS vendedor_nombre,
      p.sku, vd.descripcion,
      cat.nombre       AS categoria_nombre,
      vd.cantidad, vd.precio_unitario, vd.descuento_linea, vd.alicuota_iva,
      vd.subtotal      AS subtotal_linea,
      v.total          AS total_venta
    FROM venta v
    JOIN cliente c  ON c.id  = v.cliente_id
    JOIN usuario u  ON u.id  = v.usuario_id
    JOIN venta_detalle vd  ON vd.venta_id  = v.id
    JOIN producto p        ON p.id         = vd.producto_id
    JOIN categoria cat     ON cat.id       = p.categoria_id
    WHERE ${where}
    ORDER BY v.fecha DESC, v.numero_interno DESC, vd.id
  `).all(params) as VentaDetalleExportRow[];
}

export function registerReportesHandlers(ipcMain: IpcMain, db: Db): void {
  ipcMain.handle('reportes:detalleVentas', (_e, filtros: ReporteVentasFiltros, page = 1, pageSize = 100) => {
    const session = getSession();
    if (!session) throw new Error('UNAUTHENTICATED');
    if (!session.permisos.has('dashboard.ver')) throw new Error('FORBIDDEN');
    return detalleVentasHandler(db, filtros, page, pageSize);
  });
}

    setUsuarioId(''); setMedioPago('');
    setResultado(null); setEstado('idle'); setFiltrosActivos(undefined);
  }

  if (!can('dashboard.ver')) {
    return (
      <div style={{ padding: '2rem', textAlign: 'center' }}>
        <p className="font-bold">Sin acceso</p>
        <p className="text-muted" style={{ marginTop: '0.5rem' }}>Esta vista requiere rol Supervisor.</p>
      </div>
    );
  }

  return (
    <div style={{ display: 'flex', flexDirection: 'column', height: '100%', overflow: 'hidden' }}>

      {/* Barra de filtros */}
      <div style={{ padding: '0.75rem 1.5rem', borderBottom: '1px solid var(--line)', flexShrink: 0, display: 'flex', flexWrap: 'wrap', gap: '0.5rem', alignItems: 'flex-end' }}>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Desde</label>
          <input type="date" className="input" style={{ width: 130 }} value={fechaDesde} onChange={e => setFechaDesde(e.target.value)} />
        </div>
        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Hasta</label>
          <input type="date" className="input" style={{ width: 130 }} value={fechaHasta} onChange={e => setFechaHasta(e.target.value)} />
        </div>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Cliente</label>
          <select className="input" style={{ width: 160 }} value={clienteId} onChange={e => setClienteId(e.target.value)}>
            <option value="">Todos</option>
            {clientes.map(c => <option key={c.id} value={c.id}>{c.label}</option>)}
          </select>
        </div>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Categoría</label>
          <select className="input" style={{ width: 140 }} value={categoriaId} onChange={e => setCategoriaId(e.target.value)}>
            <option value="">Todas</option>
            {categorias.map(c => <option key={c.id} value={c.id}>{c.label}</option>)}
          </select>
        </div>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Producto</label>
          <input type="text" className="input" placeholder="SKU o nombre..." style={{ width: 150 }} value={productoTexto} onChange={e => setProductoTexto(e.target.value)} />
        </div>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Vendedor</label>
          <select className="input" style={{ width: 140 }} value={usuarioId} onChange={e => setUsuarioId(e.target.value)}>
            <option value="">Todos</option>
            {usuarios.map(u => <option key={u.id} value={u.id}>{u.label}</option>)}
          </select>
        </div>

        <div className="field" style={{ margin: 0 }}>
          <label style={{ fontSize: '0.75rem' }}>Medio de pago</label>
          <select className="input" style={{ width: 130 }} value={medioPago} onChange={e => setMedioPago(e.target.value)}>
            <option value="">Todos</option>
            <option value="efectivo">Efectivo</option>
            <option value="tarjeta">Tarjeta</option>
            <option value="transferencia">Transferencia</option>
            <option value="cuenta_corriente">Cta. Corriente</option>
            <option value="cheque">Cheque</option>
          </select>
        </div>

        <button type="button" className="btn-primary" onClick={() => buscar(1)} disabled={estado === 'loading'}>
          {estado === 'loading' ? 'Buscando…' : 'Buscar'}
        </button>
        <button type="button" className="btn-secondary" onClick={limpiar}>
          Limpiar
        </button>
      </div>

      {/* Cuerpo */}
      <div style={{ flex: 1, overflowY: 'auto', display: 'flex', flexDirection: 'column' }}>

        {estado === 'idle' && (
          <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', color: 'var(--text-muted)' }}>
            <p>Seleccioná filtros y presioná Buscar para ver resultados.</p>
          </div>
        )}

        {estado === 'error' && (
          <div style={{ padding: '2rem', textAlign: 'center' }}>
            <p className="alert-danger">Error al obtener los datos. Intentá de nuevo.</p>
          </div>
        )}

        {estado === 'empty' && (
          <div style={{ flex: 1, display: 'flex', alignItems: 'center', justifyContent: 'center', color: 'var(--text-muted)' }}>
            <p>Sin ventas para los filtros seleccionados.</p>
          </div>
        )}

        {(estado === 'results' || estado === 'loading') && resultado && (
          <>
            {/* Barra de resumen */}
            <div style={{ padding: '0.5rem 1.5rem', borderBottom: '1px solid var(--line)', display: 'flex', alignItems: 'center', gap: '1rem', flexShrink: 0 }}>
              <span style={{ fontSize: '0.875rem', color: 'var(--text-muted)' }}>
                {resultado.total} venta{resultado.total !== 1 ? 's' : ''} · Total: <strong>${fmt(resultado.totalImporte)}</strong>
              </span>
              <div style={{ marginLeft: 'auto' }}>
                <ExportarButton definicion={exportReporteVentas} filtrosActivos={filtrosActivos} />
              </div>
            </div>

            {/* Tabla */}
            <div style={{ overflowX: 'auto', flex: 1 }}>
              <table style={{ width: '100%', borderCollapse: 'collapse', fontSize: '0.875rem' }}>
                <thead>
                  <tr style={{ background: 'var(--surface-alt)', position: 'sticky', top: 0 }}>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'left', borderBottom: '1px solid var(--line)', whiteSpace: 'nowrap' }}>Nro. Venta</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'left', borderBottom: '1px solid var(--line)', whiteSpace: 'nowrap' }}>Fecha</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'left', borderBottom: '1px solid var(--line)' }}>Cliente</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'left', borderBottom: '1px solid var(--line)' }}>Vendedor</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'left', borderBottom: '1px solid var(--line)', whiteSpace: 'nowrap' }}>Medios de pago</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'right', borderBottom: '1px solid var(--line)' }}>Subtotal</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'right', borderBottom: '1px solid var(--line)' }}>IVA</th>
                    <th style={{ padding: '0.5rem 0.75rem', textAlign: 'right', borderBottom: '1px solid var(--line)' }}>Total</th>
                  </tr>
                </thead>
                <tbody>
                  {resultado.rows.map((row: ReporteVentasRow) => (
                    <tr key={row.id} style={{ borderBottom: '1px solid var(--line)' }}>
                      <td style={{ padding: '0.4rem 0.75rem', whiteSpace: 'nowrap' }}>
                        <button
                          type="button"
                          className="ia"
                          style={{ color: 'var(--accent)', textDecoration: 'underline', cursor: 'pointer', padding: 0, fontFamily: 'var(--font-mono, monospace)', fontSize: '0.875rem' }}
                          onClick={() => setVentaId(row.id)}
                        >
                          {row.numero_interno}
                        </button>
                      </td>
                      <td style={{ padding: '0.4rem 0.75rem', whiteSpace: 'nowrap' }}>{fmtFecha(row.fecha)}</td>
                      <td style={{ padding: '0.4rem 0.75rem' }}>{row.cliente_razon_social}</td>
                      <td style={{ padding: '0.4rem 0.75rem' }}>{row.vendedor_nombre}</td>
                      <td style={{ padding: '0.4rem 0.75rem', whiteSpace: 'nowrap' }}>{row.medios_pago}</td>
                      <td style={{ padding: '0.4rem 0.75rem', textAlign: 'right', fontFamily: 'var(--font-mono, monospace)' }}>${fmt(row.subtotal)}</td>
                      <td style={{ padding: '0.4rem 0.75rem', textAlign: 'right', fontFamily: 'var(--font-mono, monospace)' }}>${fmt(row.importe_iva)}</td>
                      <td style={{ padding: '0.4rem 0.75rem', textAlign: 'right', fontFamily: 'var(--font-mono, monospace)', fontWeight: 600 }}>${fmt(row.total)}</td>
                    </tr>
                  ))}
                </tbody>
              </table>
            </div>

            {/* Paginación */}
            {resultado.totalPages > 1 && (
              <div style={{ padding: '0.75rem 1.5rem', borderTop: '1px solid var(--line)', display: 'flex', alignItems: 'center', gap: '0.75rem', flexShrink: 0 }}>
                <button
                  type="button"
                  className="btn-secondary"
                  disabled={page <= 1 || estado === 'loading'}
                  onClick={() => buscar(page - 1)}
                >
                  ← Anterior
                </button>
                <span style={{ fontSize: '0.875rem', color: 'var(--text-muted)' }}>
                  Página {resultado.page} de {resultado.totalPages}
                </span>
                <button
                  type="button"
                  className="btn-secondary"
                  disabled={page >= resultado.totalPages || estado === 'loading'}
                  onClick={() => buscar(page + 1)}
                >
                  Siguiente →
                </button>
              </div>
            )}
          </>
        )}
      </div>

      {ventaId && (
        <VentaDetalleModal
          ventaId={ventaId}
          onClose={() => setVentaId(null)}
          onAnulada={() => { setVentaId(null); buscar(page); }}
        />
      )}
    </div>
  );
}
